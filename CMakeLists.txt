cmake_minimum_required(VERSION 3.16)
project(odin2_ros_driver VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(ODIN_DRIVER_NATIVE "Use local CPU tuning (never enable for portable images)" OFF)
option(ODIN_DRIVER_LTO "Enable interprocedural optimization" OFF)
option(ODIN_DRIVER_GSTREAMER "Build Jetson GStreamer hardware decoder" ON)
find_package(catkin REQUIRED COMPONENTS roscpp std_msgs sensor_msgs geometry_msgs nav_msgs
  tf2_ros tf2_msgs diagnostic_updater dynamic_reconfigure camera_info_manager message_generation)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavcodec libavutil libswscale)
if(ODIN_DRIVER_GSTREAMER)
  pkg_check_modules(GSTREAMER REQUIRED IMPORTED_TARGET gstreamer-app-1.0 gstreamer-video-1.0)
endif()
file(GLOB DRIVER_MESSAGES RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}/msg" "msg/*.msg")
file(GLOB DRIVER_SERVICES RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}/srv" "srv/*.srv")
add_message_files(FILES ${DRIVER_MESSAGES})
add_service_files(FILES ${DRIVER_SERVICES})
generate_messages(DEPENDENCIES std_msgs geometry_msgs)
generate_dynamic_reconfigure_options(cfg/Device.cfg)
catkin_package(INCLUDE_DIRS include sdk_api/include LIBRARIES odin2_driver_core odin_sdk
  CATKIN_DEPENDS roscpp std_msgs sensor_msgs geometry_msgs nav_msgs tf2_ros tf2_msgs
  diagnostic_updater dynamic_reconfigure camera_info_manager message_runtime)

# Vendor builds remain out of the source tree. EXCLUDE_FROM_ALL also prevents
# vendor installation rules from installing duplicate, architecture-named libs.
set(ODIN_BUILD_SAMPLES OFF CACHE BOOL "" FORCE)
set(ODIN_SDK_LIB_OUTPUT_DIR "${CATKIN_DEVEL_PREFIX}/lib" CACHE PATH "" FORCE)
set(ODIN_SDK_ENABLE_LTO ${ODIN_DRIVER_LTO} CACHE BOOL "" FORCE)
add_subdirectory(sdk_api ${CMAKE_CURRENT_BINARY_DIR}/vendor EXCLUDE_FROM_ALL)
include(cmake/OdinSdkStreamCompat.cmake)
add_library(odin2_driver_core src/config.cpp src/conversion.cpp src/decoder.cpp
  src/decoder_ffmpeg.cpp src/decoder_jetson.cpp src/sdk.cpp src/driver.cpp)
target_include_directories(odin2_driver_core PUBLIC include ${catkin_INCLUDE_DIRS}
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/sdk_api/include>)
target_link_libraries(odin2_driver_core PUBLIC ${catkin_LIBRARIES} odin_sdk PkgConfig::FFMPEG ${CMAKE_DL_LIBS})
if(ODIN_DRIVER_GSTREAMER)
  target_link_libraries(odin2_driver_core PRIVATE PkgConfig::GSTREAMER)
  target_compile_definitions(odin2_driver_core PRIVATE ODIN_DRIVER_GSTREAMER=1)
endif()
target_compile_options(odin2_driver_core PRIVATE -Wall -Wextra -Wpedantic)
add_dependencies(odin2_driver_core ${${PROJECT_NAME}_EXPORTED_TARGETS} ${catkin_EXPORTED_TARGETS})
if(ODIN_DRIVER_NATIVE)
  target_compile_options(odin2_driver_core PRIVATE -march=native)
endif()
if(ODIN_DRIVER_LTO)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT DRIVER_IPO_SUPPORTED)
  if(DRIVER_IPO_SUPPORTED)
    set_property(TARGET odin2_driver_core PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
  endif()
endif()
add_executable(odin2_node src/main.cpp)
target_link_libraries(odin2_node PRIVATE odin2_driver_core)
set_target_properties(odin2_node odin2_driver_core PROPERTIES INSTALL_RPATH "$ORIGIN/..;$ORIGIN")
install(TARGETS odin2_node odin2_driver_core odin_sdk
  RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION}
  LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(DIRECTORY include/${PROJECT_NAME}/ DESTINATION ${CATKIN_PACKAGE_INCLUDE_DESTINATION})
install(DIRECTORY sdk_api/include/ DESTINATION ${CATKIN_GLOBAL_INCLUDE_DESTINATION})
install(DIRECTORY config launch rviz docs DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
catkin_install_python(PROGRAMS scripts/benchmark.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  find_package(rostest REQUIRED)
  find_package(PCL REQUIRED COMPONENTS common)
  add_library(odin2_mock_sdk test/mock_sdk.cpp)
  target_link_libraries(odin2_mock_sdk PUBLIC odin2_driver_core)
  add_executable(odin2_mock_node test/mock_main.cpp)
  target_link_libraries(odin2_mock_node PRIVATE odin2_mock_sdk)
  catkin_add_gtest(driver_tests test/driver_tests.cpp test/decoder_tests.cpp)
  if(TARGET driver_tests)
    target_link_libraries(driver_tests odin2_driver_core odin2_mock_sdk)
    target_include_directories(driver_tests PRIVATE sdk_api/sdk/protocol/odinProtocol
      sdk_api/sdk/protocol/OdinProtocol sdk_api/sdk/protocol/IProtocol sdk_api/3rdparty)
  endif()
  catkin_add_gtest(cloud_pcl_tests test/cloud_pcl_tests.cpp)
  if(TARGET cloud_pcl_tests)
    target_include_directories(cloud_pcl_tests PRIVATE ${PCL_INCLUDE_DIRS})
    target_link_libraries(cloud_pcl_tests odin2_driver_core ${PCL_LIBRARIES})
  endif()
  add_rostest_gtest(runtime_tests test/runtime.test test/runtime_tests.cpp)
  if(TARGET runtime_tests)
    target_link_libraries(runtime_tests odin2_driver_core)
    add_dependencies(runtime_tests ${${PROJECT_NAME}_EXPORTED_TARGETS})
  endif()
  add_rostest_gtest(control_tests test/control.test test/control_tests.cpp)
  if(TARGET control_tests)
    target_link_libraries(control_tests odin2_driver_core)
    add_dependencies(control_tests ${${PROJECT_NAME}_EXPORTED_TARGETS} ${PROJECT_NAME}_gencfg)
  endif()
  catkin_add_gtest(sdk_stream_tests test/sdk_stream_tests.cpp
    sdk_api/3rdparty/FastCRC/FastCRCsw.cpp)
  if(TARGET sdk_stream_tests)
    target_link_libraries(sdk_stream_tests odin2_driver_core)
    target_include_directories(sdk_stream_tests PRIVATE
      sdk_api/sdk/protocol/OdinProtocol sdk_api/sdk/protocol/IProtocol sdk_api/3rdparty
      sdk_api/sdk/src sdk_api/sdk/transport/ITransport)
  endif()
  catkin_install_python(PROGRAMS test/integration.py test/async_integration.py test/startup_query.py
    DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
  add_rostest(test/integration.test DEPENDENCIES odin2_mock_node)
  add_rostest(test/async.test DEPENDENCIES odin2_mock_node)
  add_rostest(test/startup_query.test DEPENDENCIES odin2_mock_node)
endif()
